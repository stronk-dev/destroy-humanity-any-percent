type stockHookStateResearch struct {
	units, progress, remainder int64
	role                       bool
}

type stockHookDomainResearch struct {
	bundle                        CatalogBundle
	origin                        []byte
	mask                          AblationMask
	noHook                        bool
	probes, ranges, zeros, absent int
	splits                        int
}

func stockStateResearch(state *save.State) stockHookStateResearch {
	return stockHookStateResearch{units: state.StockUnits, progress: state.StockProgressMS, remainder: state.StockRateRemainderPPM}
}

func (domain *stockHookDomainResearch) hook() faction.AccrualHook {
	return faction.AccrualHook{Catalogs: faction.CatalogSet{domain.bundle.ConstantsHash: domain.bundle.Faction},
		Policies: foundationCatchupPolicies{domain.bundle.ConstantsHash: domain.bundle.Prestige.CatchupCeilingMS}}
}

// The closed faction recipe reads only elapsed time, fixed policy/catalog,
// purchased counts/mask and original stock fields. This is NOT a generic hook.
func (domain *stockHookDomainResearch) probe(at int64) (stockHookStateResearch, error) {
	state, err := save.RestoreState(domain.origin, 19, domain.bundle.Economy, economy.ScopeCompany, time.Time{})
	if err != nil {
		return stockHookStateResearch{}, err
	}
	before := stockStateResearch(state)
	if domain.noHook {
		domain.absent++
		return before, nil
	}
	if at == 0 {
		domain.zeros++
		return before, nil // Actual SimulateAdvance skips its hook on zero elapsed.
	}
	policy, err := simulationPolicyFor(domain.bundle.Economy, domain.mask)
	if err != nil {
		return stockHookStateResearch{}, err
	}
	domain.probes++
	events, hookErr := simulationHook(domain.hook(), policy).AfterAccrual(state, domain.bundle.Economy,
		save.Revision{Number: 1, ConstantsHash: domain.bundle.ConstantsHash}, accrualhook.Result{ElapsedMS: at}, nil)
	outcome := stockStateResearch(state)
	outcome.role = hasRoleActivation(policy.activations(), RoleActivation{GeneratorID: "generator.beige_tower", Kind: economy.RoleStockRate, TargetID: "faction.stock"})
	// Normalize only the declared stock write set. Encoding equality catches
	// leaked masks, changed ledger/lifetime/version/attainment or other state.
	state.StockUnits, state.StockProgressMS, state.StockRateRemainderPPM = before.units, before.progress, before.remainder
	encoded, err := save.EncodeState(state)
	if err != nil || !bytes.Equal(encoded, domain.origin) {
		return outcome, fmt.Errorf("stock hook escaped declared write set: %v", err)
	}
	if hookErr != nil {
		return outcome, hookErr
	}
	if len(events) != 0 {
		var payload struct {
			FactionID     string `json:"faction_id"`
			StockResource string `json:"stock_resource"`
			StockCap      int64  `json:"stock_cap"`
		}
		member, ok := domain.bundle.Faction.Faction(state.FactionID)
		if len(events) != 1 || events[0].Kind != save.EventFactionStockSaturated || events[0].SchemaVersion != 1 ||
			json.Unmarshal(events[0].Payload, &payload) != nil || !ok || payload.FactionID != member.ID || payload.StockResource != member.Produces || payload.StockCap != domain.bundle.Faction.StockCap {
			return outcome, fmt.Errorf("unexpected stock-hook event population")
		}
		return outcome, ErrInvalidEngineState // Existing wait refuses emitted events.
	}
	return outcome, nil
}

// Fixed purchased counts give an exact-integer stock factor. Overflow and event
// eligibility are monotone only inside the attended branch, not across its end.
func (domain *stockHookDomainResearch) certify(lo, hi int64) (bool, error) {
	if lo < 0 || hi < lo || hi > 86_460_000 {
		return false, fmt.Errorf("unsupported stock-hook research range")
	}
	domain.ranges++
	ends := []int64{hi}
	ceiling := domain.bundle.Prestige.CatchupCeilingMS
	if lo <= ceiling && hi > ceiling && !domain.noHook {
		ends = []int64{ceiling, hi}
		domain.splits++
	}
	for _, end := range ends {
		_, err := domain.probe(end)
		if errors.Is(err, ErrInvalidEngineState) || errors.Is(err, faction.ErrInvalidStockState) {
			return false, nil // Inconclusive proof, not a semantic query result.
		}
		if err != nil {
			return false, err // Instrumentation/invariant failures stay loud.
		}
	}
	return true, nil
}

func TestWaitStockHookDomainResearch(t *testing.T) {
	bundle := axisContentBundle(t)
	if bundle.Prestige.CatchupCeilingMS != 5000 || bundle.Faction.StockIntervalMS < 1500 || bundle.Faction.StockCap < 1 {
		t.Fatal("declared stock-hook fixture changed")
	}
	start := time.Date(2026, 10, 16, 12, 0, 0, 0, time.UTC)
	const horizon int64 = 6001
	total, refused, anchors, intervals := 0, 0, 0, 0
	for _, mode := range []EvaluationMode{ModeOnline, ModeOffline} {
		cashControl := make([]string, horizon+1)
		for _, row := range []struct {
			name                   string
			noHook, masked, capped bool
			firstFailure           int64
		}{
			{"no_hook", true, false, false, -1},
			{"saturating", false, false, false, 1499},
			{"masked_saturating", false, true, false, 1500},
			{"already_capped", false, false, true, -1},
		} {
			t.Run(string(mode)+"/"+row.name, func(t *testing.T) {
				initial := axisTimingState(t, bundle, start)
				initial.FactionID, initial.IncorporatedAt = "bootstrapper", start
				initial.StockUnits = bundle.Faction.StockCap - 1
				initial.StockProgressMS = bundle.Faction.StockIntervalMS - 1500
				initial.StockRateRemainderPPM = 0
				if row.capped {
					initial.StockUnits = bundle.Faction.StockCap
				}
				if err := bundle.ValidateFoundationState(initial); err != nil {
					t.Fatal("foundation rejected origin", err)
				}
				origin := mustEncodeState(t, initial)
				if _, err := save.RestoreState(origin, 19, bundle.Economy, economy.ScopeCompany, time.Time{}); err != nil {
					t.Fatal("codec rejected origin", err)
				}
				domain := stockHookDomainResearch{bundle: bundle, origin: origin, noHook: row.noHook}
				if row.masked {
					domain.mask.GeneratorIDs = []string{"generator.beige_tower"}
				}
				dependencies := SimulationDependencies{Routes: bundle.Routes}
				if !row.noHook {
					dependencies.Hook = domain.hook()
				}
				failurePrefix := make([]int, horizon+2)
				bad := 0
				for at := int64(0); at <= horizon; at++ {
					current, err := save.RestoreState(origin, 19, bundle.Economy, economy.ScopeCompany, time.Time{})
					if err != nil {
						t.Fatal(err)
					}
					result, actualErr := SimulateAdvance(current, bundle.Economy, dependencies, save.Revision{Number: 1, ConstantsHash: bundle.ConstantsHash}, mode,
						start.Add(time.Duration(at)*time.Millisecond), nil, domain.mask)
					wantFailure := row.firstFailure >= 0 && at >= row.firstFailure && at <= 5000
					if wantFailure {
						if !errors.Is(actualErr, ErrInvalidEngineState) || result.Evaluation.ElapsedMS != 0 {
							t.Fatal("wrong actual hook refusal", at, actualErr)
						}
						bad++
					} else if actualErr != nil {
						t.Fatal("valid actual wait refused", at, actualErr)
					}
					failurePrefix[at+1] = bad
					predicted, probeErr := domain.probe(at)
					observed := stockStateResearch(current)
					observed.role = hasRoleActivation(result.RoleActivations, RoleActivation{GeneratorID: "generator.beige_tower", Kind: economy.RoleStockRate, TargetID: "faction.stock"})
					if (probeErr != nil) != wantFailure || probeErr != nil && !errors.Is(probeErr, ErrInvalidEngineState) ||
						predicted.units != observed.units || predicted.progress != observed.progress || predicted.remainder != observed.remainder || !wantFailure && predicted.role != observed.role {
						t.Fatal("closed hook read set differs from full wait", at, predicted, observed, probeErr)
					}
					cash := current.Ledger.Snapshot()["company.cash"]
					if row.noHook {
						cashControl[at] = cash
					} else if row.masked && cash != "1e4" || !row.masked && cash != cashControl[at] {
						t.Fatal("stock hook unexpectedly changed actual cash", at, cash, cashControl[at])
					}
					if at == 0 || at == 1 || at == 1498 || at == 1499 || at == 1500 || at == 5000 || at == 5001 || at == horizon {
						plain, err := save.RestoreState(origin, 19, bundle.Economy, economy.ScopeCompany, time.Time{})
						if err != nil {
							t.Fatal(err)
						}
						noHook, err := SimulateAdvance(plain, bundle.Economy, SimulationDependencies{Routes: bundle.Routes}, save.Revision{Number: 1, ConstantsHash: bundle.ConstantsHash}, mode,
							start.Add(time.Duration(at)*time.Millisecond), nil, domain.mask)
						if err != nil || actualErr == nil && !reflect.DeepEqual(result.Evaluation, noHook.Evaluation) {
							t.Fatal("hook changed the underlying actual evaluation", at, err)
						}
						current.StockUnits, current.StockProgressMS, current.StockRateRemainderPPM = plain.StockUnits, plain.StockProgressMS, plain.StockRateRemainderPPM
						if !bytes.Equal(mustEncodeState(t, current), mustEncodeState(t, plain)) {
							t.Fatal("whole wait changed more than stock fields", at)
						}
						anchors++
					}
					total++
				}
				referenceProbes := domain.probes
				for _, interval := range [][2]int64{{0, 0}, {1, 1}, {1, 1498}, {1, 1499}, {1499, 1499}, {1, 1500}, {1499, 1500}, {1500, 5000}, {5000, 5000}, {5001, 5001}, {5001, horizon}, {1, horizon}, {0, horizon}} {
					safe, err := domain.certify(interval[0], interval[1])
					wantSafe := failurePrefix[interval[1]+1] == failurePrefix[interval[0]]
					if err != nil || safe != wantSafe {
						t.Errorf("hook certificate[%d,%d] safe%t want%t error%v", interval[0], interval[1], safe, wantSafe, err)
					}
					intervals++
				}
				for _, invalid := range [][2]int64{{-1, 1}, {2, 1}, {0, 86_460_001}} {
					if safe, err := domain.certify(invalid[0], invalid[1]); err == nil || safe {
						t.Fatal("invalid range was silently certified/excluded", invalid, err)
					}
				}
				if !bytes.Equal(origin, mustEncodeState(t, initial)) {
					t.Fatal("hook probe consumed origin")
				}
				if _, err := buildWaitCellsResearch(initial, bundle.Economy, mode, horizon, domain.mask, dependencies, ""); err == nil {
					t.Fatal("legacy constructor silently enabled modern/hook contexts")
				}
				refused += bad
				t.Logf("full_wait_references%d hook_refusals%d reference_helper_probes%d range_helper_probes%d ranges%d splits%d zero_skips%d absent_skips%d", horizon+1, bad, referenceProbes, domain.probes-referenceProbes, domain.ranges, domain.splits, domain.zeros, domain.absent)
			})
		}
	}
	if total != 48016 || refused != 14006 || anchors != 64 || intervals != 104 {
		t.Fatal("incomplete stock-hook population", total, refused, anchors, intervals)
	}
}
