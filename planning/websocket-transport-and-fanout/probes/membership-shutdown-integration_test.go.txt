package gameserver

import (
	"context"
	"database/sql"
	"io"
	"net/http/httptest"
	"strings"
	"testing"
	"time"

	"cloud-clicker/server/transport"

	"github.com/coder/websocket"
)

// Manual candidate proof: compiled through a Go overlay, not ordinary CI.
// Guild's guarded reader change remains unapplied pending lawful delivery.
func TestComposedDrainCancelsBlockedMembershipLookupIntegration(t *testing.T) {
	for _, scope := range []struct {
		name, table, channel, queryPattern string
		configured                         bool
	}{
		{"guild/caller", "guild_members", "guild:018f0000-0000-7000-8000-000000000fed", "SELECT EXISTS(SELECT 1 FROM guild_members WHERE account_id=$1 AND guild_id=$2 AND left_at IS NULL)", false},
		{"guild/configured", "guild_members", "guild:018f0000-0000-7000-8000-000000000fed", "SELECT EXISTS(SELECT 1 FROM guild_members WHERE account_id=$1 AND guild_id=$2 AND left_at IS NULL)", true},
		{"cohort/caller", "founder_commons_assignments", "cohort:018f0000-0000-7000-8000-000000000fed", "%WHERE assignment.founder_id::text=$1 AND assignment.cohort_id::text=$2%", false},
		{"cohort/configured", "founder_commons_assignments", "cohort:018f0000-0000-7000-8000-000000000fed", "%WHERE assignment.founder_id::text=$1 AND assignment.cohort_id::text=$2%", true},
	} {
		t.Run(scope.name, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), 40*time.Second)
			defer cancel()
			database := newCredentialCleanupDatabase(t, ctx)
			defer database.Close()
			clock := &mutableClock{now: time.Now().UTC()}
			_, composition := newCredentialCleanupComposition(t, ctx, database, clock, io.Discard)
			serverCtx, stopServer := context.WithCancel(ctx)
			defer stopServer()
			if err := composition.Server.Start(serverCtx); err != nil {
				t.Fatal(err)
			}
			account, err := composition.Accounts.CreateAccount(ctx)
			if err != nil {
				t.Fatal(err)
			}
			pair, err := composition.Accounts.CreateSession(ctx, account.AccountID, account.RecoveryCode)
			if err != nil {
				t.Fatal(err)
			}
			httpServer := httptest.NewServer(composition.Server.Handler())
			defer httpServer.Close()
			peer := dialCompositionSocket(t, "ws"+strings.TrimPrefix(httpServer.URL, "http")+"/connection/websocket", httpServer.Client(), pair.AccessToken)
			defer peer.connection.CloseNow()
			writeWS(t, peer, map[string]any{"id": 2, "subscribe": map[string]any{"channel": "world"}})
			if reply := readWSID(t, peer, 2); reply.Error != nil || reply.Subscribe == nil {
				t.Fatal("healthy public subscription failed")
			}
			// These are real nonmember queries, not seeded membership or a
			// claim that the fixture proves natural Guild/Commons progression.
			writeWS(t, peer, map[string]any{"id": 3, "subscribe": map[string]any{"channel": scope.channel}})
			if reply := readWSID(t, peer, 3); reply.Error == nil || reply.Error.Code != 103 {
				t.Fatal("nonmember subscription was not denied")
			}
			closed := make(chan websocket.StatusCode, 1)
			go func() { closed <- waitSocketCloseStatus(t, peer.connection, 35*time.Second) }()
			lock, err := database.BeginTx(ctx, nil)
			if err != nil {
				t.Fatal(err)
			}
			defer lock.Rollback()
			// The identifier is from the closed literal population above.
			if _, err := lock.ExecContext(ctx, "LOCK TABLE "+scope.table+" IN ACCESS EXCLUSIVE MODE"); err != nil {
				t.Fatal(err)
			}
			writeWS(t, peer, map[string]any{"id": 4, "subscribe": map[string]any{"channel": scope.channel}})
			var permissionPID int
			for {
				err := database.QueryRowContext(ctx, `SELECT a.pid FROM pg_stat_activity a JOIN pg_locks l USING(pid)
					WHERE a.datname=current_database() AND l.relation=$1::regclass
					AND NOT l.granted AND l.mode='AccessShareLock' AND a.query LIKE $2`, scope.table, scope.queryPattern).Scan(&permissionPID)
				if err == nil {
					break
				}
				if err != sql.ErrNoRows {
					t.Fatal(err)
				}
				select {
				case <-ctx.Done():
					t.Fatal("actual membership query never blocked")
				case status := <-closed:
					t.Fatalf("healthy socket closed before membership query: %d", status)
				case <-time.After(10 * time.Millisecond):
				}
			}
			drainBound := 2 * time.Second
			if scope.configured {
				drainBound = composition.Node.DrainTimeout()
				if drainBound != 15*time.Second {
					t.Fatal("configured drain policy changed; reconcile observation")
				}
			}
			drainCtx, stopDrain := context.WithTimeout(ctx, drainBound)
			defer stopDrain()
			if err := composition.Server.Drain(drainCtx, clock.Time()); err != nil {
				t.Fatalf("drain with blocked %s membership query: %v", scope.name, err)
			}
			select {
			case status := <-closed:
				if status != websocket.StatusCode(transport.CloseServerDrain) {
					t.Fatalf("membership drain close=%d want=%d", status, transport.CloseServerDrain)
				}
			case <-drainCtx.Done():
				t.Fatal("socket survived completed membership drain")
			}
			t.Log("drain returned success and socket closed4003; lock remains held")
			// Keep the exclusive lock and database live until cancellation is
			// observed. Closing the fixture must not supply the pass condition.
			reportedWait := false
			for {
				var waiting bool
				if err := database.QueryRowContext(drainCtx, "SELECT EXISTS(SELECT 1 FROM pg_locks WHERE pid=$1 AND NOT granted)", permissionPID).Scan(&waiting); err != nil {
					t.Fatal(err)
				}
				if !waiting {
					break
				}
				if !reportedWait {
					t.Log("owner SQL still blocked after successful drain")
					reportedWait = true
				}
				select {
				case <-drainCtx.Done():
					t.Fatal("membership query survived drain")
				case <-time.After(10 * time.Millisecond):
				}
			}
		})
	}
}
