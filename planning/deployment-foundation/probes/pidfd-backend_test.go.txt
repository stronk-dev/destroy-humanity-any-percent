type nativeBinding struct {
	PID    int    `json:"pid"`
	Start  string `json:"start"`
	Inode  string `json:"inode"`
	Local  string `json:"local"`
	Remote string `json:"remote"`
	Parent int    `json:"parent"`
}

type nativeReply struct {
	Sequence   int           `json:"sequence"`
	Outcome    string        `json:"outcome"`
	Code       string        `json:"code"`
	Signals    int           `json:"signals_sent"`
	NativeExit bool          `json:"native_exit"`
	Errno      int           `json:"errno"`
	Subjects   int           `json:"subjects"`
	Binding    nativeBinding `json:"binding"`
	SHA        string        `json:"binding_sha256"`
}

type nativeRPC struct {
	mu         sync.Mutex
	directory  string
	sequence   int
	transcript []nativeReply
}

func (m *nativeRPC) call(ctx context.Context, request map[string]any) (nativeReply, error) {
	ctx, stop := context.WithTimeout(ctx, 10*time.Second)
	defer stop()
	if err := researchDeadline(ctx); err != nil {
		return nativeReply{}, err
	}
	// The budget includes waiting for serialization. A cancelled caller may
	// not wait behind an unbounded mutex or publish a native command later.
	retry := time.NewTicker(time.Millisecond)
	defer retry.Stop()
	for !m.mu.TryLock() {
		select {
		case <-ctx.Done():
			return nativeReply{}, ctx.Err()
		case <-retry.C:
		}
	}
	defer m.mu.Unlock()
	if err := researchDeadline(ctx); err != nil {
		return nativeReply{}, err
	}
	m.sequence++
	path := filepath.Join(m.directory, fmt.Sprintf("request-%06d.json", m.sequence))
	data, err := json.Marshal(request)
	if err != nil {
		return nativeReply{}, err
	}
	if err := researchDeadline(ctx); err != nil {
		return nativeReply{}, err
	}
	file, err := os.OpenFile(path+".temporary", os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0644)
	if err != nil {
		return nativeReply{}, err
	}
	err = researchWrite(file, data)
	if err == nil {
		err = file.Sync()
	}
	if closeErr := file.Close(); err == nil {
		err = closeErr
	}
	if err != nil {
		return nativeReply{}, err
	}
	if err := researchDeadline(ctx); err != nil {
		return nativeReply{}, err
	}
	if err = os.Rename(path+".temporary", path); err != nil {
		return nativeReply{}, err
	}
	response := filepath.Join(m.directory, fmt.Sprintf("response-%06d.json", m.sequence))
	for {
		data, err = os.ReadFile(response)
		if errors.Is(err, os.ErrNotExist) {
			if err := researchDeadline(ctx); err != nil {
				return nativeReply{}, err
			}
			time.Sleep(5 * time.Millisecond)
			continue
		}
		if err != nil {
			return nativeReply{}, err
		}
		var reply nativeReply
		if err := json.Unmarshal(data, &reply); err != nil {
			return nativeReply{}, err
		}
		if err := researchDeadline(ctx); err != nil {
			return nativeReply{}, err
		}
		if reply.Sequence != m.sequence {
			return nativeReply{}, errors.New("observer response sequence: INVALID")
		}
		m.transcript = append(m.transcript, reply)
		return reply, nil
	}
}

type opaqueSession struct {
	name          string
	front, back   *net.TCPConn
	bound         nativeReply
	forward       chan bool
	once          sync.Once
	frontTerminal bool
	frontEnd      string
	serverEnd     string
	nativeExit    bool
}

type opaqueRelay struct {
	mu                   sync.Mutex
	listener             *net.TCPListener
	sessions             []*opaqueSession
	closed, killExpected bool
	err                  error
	wg                   sync.WaitGroup
	ctx                  context.Context
	end                  time.Time
	monitor              *nativeRPC
	address, prefix      string
	automatic            bool
}

func newOpaqueRelay(t *testing.T, ctx context.Context, m *nativeRPC, address, prefix string, automatic bool) *opaqueRelay {
	t.Helper()
	listener, err := net.ListenTCP("tcp4", &net.TCPAddr{IP: net.IPv4(127, 0, 0, 1)})
	if err != nil {
		t.Fatal(err)
	}
	end, ok := ctx.Deadline()
	if !ok {
		t.Fatal("missing population deadline")
	}
	if err := listener.SetDeadline(end); err != nil {
		t.Fatal(err)
	}
	r := &opaqueRelay{listener: listener, ctx: ctx, end: end, monitor: m, address: address, prefix: prefix, automatic: automatic}
	r.wg.Add(1)
	go func() {
		defer r.wg.Done()
		for {
			front, err := listener.AcceptTCP()
			if err != nil {
				r.mu.Lock()
				if !r.closed {
					r.err = err
				}
				r.mu.Unlock()
				return
			}
			r.mu.Lock()
			if r.closed {
				r.mu.Unlock()
				front.Close()
				continue
			}
			s := &opaqueSession{name: fmt.Sprintf("%s-%d", prefix, len(r.sessions)+1), front: front, forward: make(chan bool, 1)}
			r.sessions = append(r.sessions, s)
			r.wg.Add(1)
			r.mu.Unlock()
			go r.serve(s)
		}
	}()
	t.Cleanup(func() {
		r.closeAdmission()
		r.mu.Lock()
		for _, s := range r.sessions {
			s.once.Do(func() { s.forward <- false })
			s.front.Close()
			if s.back != nil {
				s.back.Close()
			}
		}
		r.mu.Unlock()
		r.wg.Wait()
	})
	return r
}

func (r *opaqueRelay) fail(err error) {
	r.mu.Lock()
	if r.err == nil {
		r.err = err
	}
	r.mu.Unlock()
}
func (r *opaqueRelay) closeAdmission() {
	r.mu.Lock()
	r.closed = true
	r.mu.Unlock()
	r.listener.Close()
}

func (r *opaqueRelay) serve(s *opaqueSession) {
	defer r.wg.Done()
	defer s.front.Close()
	dial, stop := context.WithTimeout(r.ctx, 10*time.Second)
	conn, err := (&net.Dialer{}).DialContext(dial, "tcp4", r.address)
	if late := researchDeadline(dial); err == nil && late != nil {
		conn.Close()
		err = late
	}
	stop()
	if err != nil {
		r.fail(err)
		return
	}
	back := conn.(*net.TCPConn)
	defer back.Close()
	r.mu.Lock()
	s.back = back
	r.mu.Unlock()
	if err := back.SetDeadline(r.end); err != nil {
		r.fail(err)
		return
	}
	if err := s.front.SetDeadline(r.end); err != nil {
		r.fail(err)
		return
	}
	// The trusted relay supplies its ACTUAL original socket endpoints. No
	// client bytes, including the SSLRequest, are forwarded before binding.
	bound, err := r.monitor.call(r.ctx, map[string]any{"action": "register", "subject": s.name, "local": back.RemoteAddr().String(), "remote": back.LocalAddr().String()})
	if err != nil {
		r.fail(err)
		return
	}
	if bound.Outcome != "bound" || bound.Binding.PID <= 1 || bound.Binding.Parent != 1 || len(bound.SHA) != 64 || bound.Binding.Local != back.RemoteAddr().String() || bound.Binding.Remote != back.LocalAddr().String() {
		r.fail(errors.New("native binding mismatch: INVALID"))
		return
	}
	r.mu.Lock()
	s.bound = bound
	r.mu.Unlock()
	if r.automatic {
		s.once.Do(func() { s.forward <- true })
	}
	frontDone := make(chan error, 1)
	go func() {
		var allow bool
		select {
		case allow = <-s.forward:
		case <-r.ctx.Done():
			frontDone <- r.ctx.Err()
			return
		}
		var target io.Writer = io.Discard
		if allow {
			target = back
		}
		_, err := io.Copy(target, s.front)
		end := "peer_eof"
		if errors.Is(err, net.ErrClosed) {
			end = "locally_closed"
			err = nil
		}
		if errors.Is(err, syscall.ECONNRESET) {
			end = "reset"
			err = nil
		}
		if errors.Is(err, syscall.EPIPE) {
			end = "broken_pipe"
			err = nil
		}
		if err == nil && allow {
			if closeErr := back.CloseWrite(); closeErr != nil && !errors.Is(closeErr, net.ErrClosed) && !errors.Is(closeErr, syscall.ENOTCONN) && !errors.Is(closeErr, syscall.EPIPE) && !errors.Is(closeErr, syscall.ECONNRESET) {
				err = closeErr
			}
		}
		r.mu.Lock()
		if err == nil {
			s.frontTerminal = true
			s.frontEnd = end
		}
		r.mu.Unlock()
		frontDone <- err
	}()
	buffer := make([]byte, 32<<10)
	serverErr := error(nil)
	downstreamLost := false
	for {
		n, err := back.Read(buffer)
		if n > 0 && !downstreamLost {
			if writeErr := researchWrite(s.front, buffer[:n]); writeErr != nil {
				if errors.Is(writeErr, syscall.EPIPE) || errors.Is(writeErr, syscall.ECONNRESET) {
					downstreamLost = true
				} else {
					serverErr = writeErr
					break
				}
			}
		}
		if errors.Is(err, io.EOF) {
			r.mu.Lock()
			s.serverEnd = "eof"
			r.mu.Unlock()
			break
		}
		if errors.Is(err, syscall.ECONNRESET) {
			r.mu.Lock()
			s.serverEnd = "reset"
			r.mu.Unlock()
			break
		}
		if err != nil {
			serverErr = err
			break
		}
	}
	// End our forwarding actor. This local close is NOT native completion;
	// terminal() separately requires the retained original backend pidfd exit.
	s.front.Close()
	s.once.Do(func() { s.forward <- false })
	if serverErr != nil {
		back.Close()
	}
	if err := <-frontDone; serverErr == nil {
		serverErr = err
	}
	if serverErr != nil {
		r.fail(serverErr)
	}
}

func (r *opaqueRelay) terminal() (bool, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.err != nil {
		return false, r.err
	}
	if !r.closed || len(r.sessions) == 0 {
		return false, nil
	}
	for _, s := range r.sessions {
		if s.bound.Outcome != "bound" || !s.frontTerminal || (s.serverEnd != "eof" && s.serverEnd != "reset") || !s.nativeExit {
			return false, nil
		}
	}
	return true, nil
}

func (r *opaqueRelay) collectNativeExits(t *testing.T, ctx context.Context) {
	t.Helper()
	r.mu.Lock()
	sessions := append([]*opaqueSession(nil), r.sessions...)
	r.mu.Unlock()
	for _, s := range sessions {
		r.mu.Lock()
		bound := s.bound
		r.mu.Unlock()
		if bound.Outcome != "bound" {
			t.Fatal("unbound original session: INVALID")
		}
		reply := nativeCall(t, ctx, r.monitor, map[string]any{"action": "observe_exit", "subject": s.name, "binding_sha256": bound.SHA}, "exited")
		if !reply.NativeExit {
			t.Fatal("missing retained original pidfd exit")
		}
		r.mu.Lock()
		s.nativeExit = true
		r.mu.Unlock()
	}
}

func (r *opaqueRelay) report(t *testing.T, minimum int) []map[string]any {
	t.Helper()
	complete, err := r.terminal()
	if err != nil || !complete {
		t.Fatal("incomplete opaque relay: INVALID", err)
	}
	r.wg.Wait()
	r.mu.Lock()
	defer r.mu.Unlock()
	if len(r.sessions) < minimum {
		t.Fatal("missing real helper connections")
	}
	result := make([]map[string]any, 0, len(r.sessions))
	for _, s := range r.sessions {
		result = append(result, map[string]any{"binding": s.bound, "front_end": s.frontEnd, "server_end": s.serverEnd, "native_exit": s.nativeExit})
	}
	return result
}

func nativeCall(t *testing.T, ctx context.Context, m *nativeRPC, request map[string]any, outcome string) nativeReply {
	t.Helper()
	reply, err := m.call(ctx, request)
	if err != nil || reply.Outcome != outcome {
		t.Fatal("native observation failed: INVALID", reply.Outcome, reply.Code, err)
	}
	return reply
}

	termination := nativeCall(t, interrupted, monitor, map[string]any{"action": "terminate", "subject": blocked.name, "binding_sha256": blocked.bound.SHA}, "terminated")
	if !termination.NativeExit || termination.Signals != 1 {
		t.Fatal("native original process exit missing")
	}
	r.collectNativeExits(t, interrupted)
	researchPoll(t, interrupted, r.terminal)
	dead := nativeCall(t, interrupted, monitor, map[string]any{"action": "dead_reference", "subject": blocked.name, "binding_sha256": blocked.bound.SHA}, "old_reference_dead")
	if dead.Errno != int(syscall.ESRCH) {
		t.Fatal("old pidfd did not retain exited subject")
	}
	if _, err := control.ExecContext(interrupted, "COMMIT"); err != nil {
		t.Fatal(err)
	}
	interruptedResult := map[string]any{"bindings": r.report(t, 2), "wrong_binding": wrong, "termination": termination, "old_reference": dead, "all_native_and_stream_terminal_before_unlock": true}
	if researchDeadline(interrupted) != nil {
		t.Fatal("interrupted population deadline: INVALID")
	}
	stopInterrupted()

	startup, stopStartup := context.WithTimeout(context.Background(), 60*time.Second)
	defer stopStartup()
	sr := newOpaqueRelay(t, startup, monitor, "postgres:5432", "startup", false)
	config.Directory = t.TempDir()
	config.URLFile = writeSecret(t, work, "startup-gateway", withTLS(sourceURL, sr.listener.Addr().String()))
	config.Report = filepath.Join(work, "startup-report.json")
	child, done = researchChild(t, config)
	var pending *opaqueSession
	researchPoll(t, startup, func() (bool, error) {
		sr.mu.Lock()
		defer sr.mu.Unlock()
		if sr.err != nil {
			return false, sr.err
		}
		if len(sr.sessions) == 1 && sr.sessions[0].bound.Outcome == "bound" {
			pending = sr.sessions[0]
			return true, nil
		}
		return false, nil
	})
	if researchScalar(t, startup, observer, "SELECT count(*)::text FROM pg_stat_activity WHERE pid=$1", pending.bound.Binding.PID) != "0" {
		t.Fatal("controlled startup is not absent from activity rows")
	}
	sr.closeAdmission()
	sr.mu.Lock()
	sr.killExpected = true
	sr.mu.Unlock()
	if err := syscall.Kill(-child.Process.Pid, syscall.SIGKILL); err != nil {
		t.Fatal(err)
	}
	researchWaitChild(t, startup, done, false)
	startupTermination := nativeCall(t, startup, monitor, map[string]any{"action": "terminate", "subject": pending.name, "binding_sha256": pending.bound.SHA}, "terminated")
	if !startupTermination.NativeExit || startupTermination.Signals != 2 {
		t.Fatal("startup original native exit missing")
	}
	pending.once.Do(func() { pending.forward <- false }) // Drain client bytes; never forward after closure.
	sr.collectNativeExits(t, startup)
	researchPoll(t, startup, sr.terminal)
	startupDead := nativeCall(t, startup, monitor, map[string]any{"action": "dead_reference", "subject": pending.name, "binding_sha256": pending.bound.SHA}, "old_reference_dead")
	if startupDead.Errno != int(syscall.ESRCH) {
		t.Fatal("startup dead reference lost binding")
	}
	if researchScalar(t, startup, control, "SELECT pg_backend_pid()::text") != controlPID || researchScalar(t, startup, control, "SELECT pg_postmaster_start_time()::text") != postmaster {
		t.Fatal("bystander/postmaster replaced")
	}
	if researchScalar(t, startup, control, "SELECT string_agg(id::text,',' ORDER BY id) FROM public.native_pidfd_probe") != "7" {
		t.Fatal("prior row changed")
	}
	if _, err := control.ExecContext(startup, "INSERT INTO public.native_pidfd_probe VALUES(42)"); err != nil {
		t.Fatal(err)
	}
	rows := researchScalar(t, startup, control, "SELECT string_agg(id::text,',' ORDER BY id) FROM public.native_pidfd_probe")
	if rows != "7,42" {
		t.Fatal("continuous control unusable")
	}
	for _, path := range []string{filepath.Join(work, "interrupted-report.json"), config.Report} {
		if _, err := os.Stat(path); !errors.Is(err, os.ErrNotExist) {
			t.Fatal("killed helper published completion")
		}
	}
	startupResult := map[string]any{"bindings": sr.report(t, 1), "activity_rows": 0, "termination": startupTermination, "old_reference": startupDead, "original_stream_terminal": true}
	stopped := nativeCall(t, startup, monitor, map[string]any{"action": "stop"}, "stopped")
	if stopped.Subjects != 7 || stopped.Signals != 2 {
		t.Fatal("observer population incomplete")
	}
	ready, err := os.ReadFile(filepath.Join(monitor.directory, "observer-ready.json"))
	if err != nil {
		t.Fatal(err)
	}
	var nativeProfile any
	if json.Unmarshal(ready, &nativeProfile) != nil {
		t.Fatal("invalid native profile")
	}
	result := map[string]any{"objective_completed": true, "positive": positiveResult, "interrupted": interruptedResult, "startup": startupResult, "observer": nativeProfile, "transcript": monitor.transcript, "rows": rows, "same_bystander_and_postmaster": true, "protocol_key_parsing": false, "tls_termination": false, "population_work_guard_seconds": 60, "handshake_seconds": 10}
	data, err = json.Marshal(result)
	if err != nil || researchDeadline(startup) != nil {
		t.Fatal("final collection invalid")
	}
	if err := os.WriteFile(filepath.Join(monitor.directory, "result.json"), data, 0600); err != nil {
		t.Fatal(err)
	}
	if researchDeadline(startup) != nil {
		t.Fatal("report write late: INVALID")
	}
	t.Log("real TLS backup/restore, interrupted original and pre-activity startup native ownership complete")
}

func TestManualPIDfdTerminalControls(t *testing.T) {
	for _, c := range []struct {
		name, server, binding           string
		closed, front, native, complete bool
		err                             error
	}{
		{"reset alone", "reset", "bound", true, true, false, false, nil},
		{"EOF alone", "eof", "bound", true, true, false, false, nil},
		{"native exit alone", "", "bound", true, true, true, false, nil},
		{"unknown stream error", "timeout", "bound", true, true, true, false, nil},
		{"forwarder unfinished", "reset", "bound", true, false, true, false, nil},
		{"unbound subject", "eof", "", true, true, true, false, nil},
		{"admission open", "eof", "bound", false, true, true, false, nil},
		{"native EOF completion", "eof", "bound", true, true, true, true, nil},
		{"native reset completion", "reset", "bound", true, true, true, true, nil},
		{"permission error", "reset", "bound", true, true, true, false, os.ErrPermission},
		{"deadline error", "reset", "bound", true, true, true, false, context.DeadlineExceeded},
	} {
		t.Run(c.name, func(t *testing.T) {
			r := &opaqueRelay{closed: c.closed, err: c.err, sessions: []*opaqueSession{{bound: nativeReply{Outcome: c.binding}, frontTerminal: c.front, serverEnd: c.server, nativeExit: c.native}}}
			complete, err := r.terminal()
			if complete != c.complete || !errors.Is(err, c.err) {
				t.Fatal("native terminal producer accepted incomplete/error work or refused complete work")
			}
		})
	}
}

type nativeCancelJSON struct{ cancel context.CancelFunc }

func (c nativeCancelJSON) MarshalJSON() ([]byte, error) {
	c.cancel()
	return []byte(`"cancelled"`), nil
}

func TestManualPIDfdRPCAdmissionControls(t *testing.T) {
	noPublication := func(t *testing.T, m *nativeRPC) {
		t.Helper()
		entries, err := os.ReadDir(m.directory)
		if err != nil {
			t.Fatal(err)
		}
		for _, entry := range entries {
			if strings.HasPrefix(entry.Name(), "request-") && !strings.HasSuffix(entry.Name(), ".temporary") {
				t.Fatal("expired/cancelled call published a native command")
			}
		}
	}
	for _, expired := range []bool{false, true} {
		name := "already cancelled"
		if expired {
			name = "already expired"
		}
		t.Run(name, func(t *testing.T) {
			m := &nativeRPC{directory: t.TempDir()}
			var ctx context.Context
			var stop context.CancelFunc
			if expired {
				ctx, stop = context.WithDeadline(context.Background(), time.Now().Add(-time.Second))
			} else {
				ctx, stop = context.WithCancel(context.Background())
				stop()
			}
			defer stop()
			_, err := m.call(ctx, map[string]any{"action": "unit"})
			expect := context.Canceled
			if expired {
				expect = context.DeadlineExceeded
			}
			if !errors.Is(err, expect) || m.sequence != 0 {
				t.Fatal("already invalid call admitted", err)
			}
			noPublication(t, m)
		})
	}
	t.Run("cancel while encoding", func(t *testing.T) {
		m := &nativeRPC{directory: t.TempDir()}
		ctx, stop := context.WithCancel(context.Background())
		defer stop()
		_, err := m.call(ctx, map[string]any{"action": nativeCancelJSON{cancel: stop}})
		if !errors.Is(err, context.Canceled) {
			t.Fatal("encoding cancellation not refused", err)
		}
		noPublication(t, m)
	})
	t.Run("queued expiry before unlock", func(t *testing.T) {
		m := &nativeRPC{directory: t.TempDir()}
		m.mu.Lock()
		ctx, stop := context.WithTimeout(context.Background(), 25*time.Millisecond)
		defer stop()
		done := make(chan error, 1)
		go func() { _, err := m.call(ctx, map[string]any{"action": "unit"}); done <- err }()
		select {
		case err := <-done:
			m.mu.Unlock()
			if !errors.Is(err, context.DeadlineExceeded) {
				t.Fatal("queued expiry not refused", err)
			}
		case <-time.After(time.Second):
			m.mu.Unlock()
			<-done
			t.Fatal("RPC waited for mutex after caller expired")
		}
		if m.sequence != 0 {
			t.Fatal("expired queue entry allocated a command")
		}
		noPublication(t, m)
	})
	t.Run("queued timely reply", func(t *testing.T) {
		m := &nativeRPC{directory: t.TempDir()}
		if err := os.WriteFile(filepath.Join(m.directory, "response-000001.json"), []byte(`{"sequence":1,"outcome":"unit_reply"}`), 0600); err != nil {
			t.Fatal(err)
		}
		m.mu.Lock()
		ctx, stop := context.WithTimeout(context.Background(), time.Second)
		defer stop()
		done := make(chan error, 1)
		go func() {
			reply, err := m.call(ctx, map[string]any{"action": "unit"})
			if err == nil && reply.Outcome != "unit_reply" {
				err = errors.New("reply mismatch")
			}
			done <- err
		}()
		m.mu.Unlock()
		select {
		case err := <-done:
			if err != nil {
				t.Fatal(err)
			}
		case <-ctx.Done():
			t.Fatal("timely queued call did not complete")
		}
		if _, err := os.Stat(filepath.Join(m.directory, "request-000001.json")); err != nil {
			t.Fatal("timely command not published", err)
		}
	})
	t.Run("write error is not refusal", func(t *testing.T) {
		path := filepath.Join(t.TempDir(), "not-a-directory")
		if err := os.WriteFile(path, []byte("file"), 0600); err != nil {
			t.Fatal(err)
		}
		m := &nativeRPC{directory: path}
		ctx, stop := context.WithTimeout(context.Background(), time.Second)
		defer stop()
		reply, err := m.call(ctx, map[string]any{"action": "unit"})
		if err == nil || reply.Outcome != "" {
			t.Fatal("native request write failure became an outcome")
		}
	})
}
