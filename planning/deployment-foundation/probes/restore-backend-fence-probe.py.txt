"""Manual CR4 research. Disposable declared Postgres fixture, never production.

Run from the repository root with Python 3. This does not implement a recovery
policy or prove all Engine/Compose request lifetimes. Guard exhaustion invalidates
the observation. The three outcomes are recorded, not forced to a hypothesis.
"""
import json
import pathlib
import queue
import secrets
import shutil
import subprocess
import tempfile
import threading
import time


def main():
    root = pathlib.Path.cwd()
    assert (root / "compose.deployment-backup-test.yml").is_file()
    # Exact linux/amd64 child of the declared cf78e766... index. Select it
    # explicitly instead of replacing that index's cached ARM64 identity.
    image = "postgres:16-alpine@sha256:075f7ba66bc9b3ce7d6b8b635208ff61cd7cf1a67d71ec530eec5d7ae0cbe571"
    nonce = secrets.token_hex(8)
    project = "cc-restore-fence-" + nonce
    work = pathlib.Path(tempfile.mkdtemp(prefix="cc-restore-fence-"))
    override = work / "compose.yml"
    override.write_text(
        "services:\n  postgres:\n    image: " + image + "\n"
        "  restore-probe:\n    image: " + image + "\n"
        "    platform: linux/amd64\n    read_only: true\n"
        "    entrypoint: [pg_restore]\n    environment:\n"
        "      PGHOST: postgres\n      PGUSER: cloud_clicker\n"
        "      PGPASSWORD: cloud_clicker_test\n    volumes:\n"
        "      - type: bind\n        source: " + json.dumps(str(work)) + "\n"
        "        target: /probe\n        read_only: true\n"
    )
    compose = ["compose", "--project-name", project, "-f",
               str(root / "compose.deployment-backup-test.yml"), "-f", str(override)]
    deadline = time.monotonic() + 60
    lock_process = None

    def run(*args, cleanup=False):
        timeout = 15 if cleanup else min(10, deadline - time.monotonic())
        if timeout <= 0:
            raise RuntimeError("whole-experiment guard exhausted: INVALID")
        result = subprocess.run(["docker", *args], text=True, capture_output=True,
                                timeout=timeout)
        if result.returncode != 0:
            raise RuntimeError("fixture Docker command failed: " + result.stdout + result.stderr)
        return result.stdout.strip()

    def owned(container, service):
        assert len(container) == 64 and all(c in "0123456789abcdef" for c in container)
        fields = run("inspect", "--format", '{{.Id}} {{index .Config.Labels "com.docker.compose.project"}} {{index .Config.Labels "com.docker.compose.service"}}', container)
        assert fields == container + " " + project + " " + service, "unowned container"
        assert run("inspect", "--format", "{{.Image}}", container) == "sha256:75f5a96988cdf694a215073c3e9c001b706b371e2f94df3967f2efdec2787f6b", "wrong fixture image config"

    def wait_for(check):
        until = min(deadline, time.monotonic() + 10)
        while True:
            value = check()
            if value:
                return value
            if time.monotonic() >= until:
                raise RuntimeError("handshake guard exhausted: INVALID")
            time.sleep(0.01)

    observations = []
    try:
        run(*compose, "up", "--detach", "--wait", "postgres")
        database_id = run(*compose, "ps", "--quiet", "postgres")
        owned(database_id, "postgres")

        def sql(database, query):
            return run("exec", database_id, "psql", "-U", "cloud_clicker", "-d",
                       database, "-Atq", "-v", "ON_ERROR_STOP=1", "-c", query)

        sql("cloud_clicker_source", "CREATE FUNCTION public.restore_fence_check(integer) RETURNS boolean LANGUAGE plpgsql AS $$ BEGIN PERFORM pg_advisory_xact_lock(743831); RETURN true; END $$; CREATE TABLE public.restore_fence_probe(id integer CHECK(public.restore_fence_check(id))); INSERT INTO public.restore_fence_probe VALUES(7)")
        run("exec", database_id, "pg_dump", "-U", "cloud_clicker", "-d", "cloud_clicker_source",
            "--format=custom", "--no-owner", "--no-privileges", "--file=/tmp/fence.dump")
        run("cp", database_id + ":/tmp/fence.dump", str(work / "fence.dump"))

        for arm in ("normal", "remove_client", "remove_client_and_backend"):
            database = "restore_fence_" + arm
            application = "cc_restore_" + nonce + "_" + arm
            sql("postgres", "CREATE DATABASE " + database + " TEMPLATE template0")
            # No extra target object or trigger is installed. The custom dump's
            # synthetic CHECK function holds COPY at the actual backend gate.
            args = ["docker", "exec", "-i", database_id, "psql", "-U", "cloud_clicker",
                    "-d", database, "-Atq", "-v", "ON_ERROR_STOP=1"]
            lock_process = subprocess.Popen(args, text=True, stdin=subprocess.PIPE,
                                            stdout=subprocess.PIPE, stderr=subprocess.PIPE)
            ready = queue.Queue()
            def read_ready(process=lock_process):
                ready.put(process.stdout.readline().strip())
            reader = threading.Thread(target=read_ready, daemon=True)
            reader.start()
            lock_process.stdin.write("BEGIN; DO $$ BEGIN PERFORM pg_advisory_xact_lock(743831); END $$; SELECT 'lock-ready';\n")
            lock_process.stdin.flush()
            assert ready.get(timeout=10) == "lock-ready", "lock setup failed: INVALID"

            name = project + "-" + arm
            container = run(*compose, "run", "--detach", "--no-deps", "--name", name,
                            "-e", "PGAPPNAME=" + application, "restore-probe", "--exit-on-error",
                            "--no-owner", "--no-privileges", "--dbname=" + database, "/probe/fence.dump")
            owned(container, "restore-probe")
            predicate = "datname='" + database + "' AND application_name='" + application + "'"
            last_backend = {}
            def blocked_backend():
                text = sql("postgres", "SELECT json_build_object('pid',pid,'state',state,'wait',wait_event_type,'wait_event',wait_event,'query',query) FROM pg_stat_activity WHERE " + predicate)
                if not text:
                    return None
                row = json.loads(text)
                last_backend.clear()
                last_backend.update(row)
                return row if row["state"] == "active" and row["wait"] == "Lock" and row["wait_event"] == "advisory" and row["query"].lstrip().startswith("COPY public.restore_fence_probe") else None
            try:
                backend = wait_for(blocked_backend)
            except RuntimeError:
                print(json.dumps(dict(objective_completed=False, guard_exhausted=True,
                                      arm=arm, last_backend=last_backend)), flush=True)
                raise
            baseline_rows = int(sql(database, "SELECT count(*) FROM public.restore_fence_probe"))
            assert baseline_rows == 0, "restore row committed before its gate: INVALID"
            backend_after_removal = None
            terminated = None
            if arm != "normal":
                owned(container, "restore-probe")
                run("rm", "--force", "--volumes", container)
                assert run("ps", "--all", "--quiet", "--no-trunc", "--filter", "id=" + container) == ""
                backend_after_removal = sql("postgres", "SELECT count(*) FROM pg_stat_activity WHERE " + predicate + " AND pid=" + str(backend["pid"])) == "1"
                if arm == "remove_client_and_backend":
                    # Exact observed PID plus unique database/application; never
                    # terminate arbitrary sessions or another operation's role.
                    terminated = sql("postgres", "SELECT pg_terminate_backend(pid,1000) FROM pg_stat_activity WHERE " + predicate + " AND pid=" + str(backend["pid"]))
                    wait_for(lambda: sql("postgres", "SELECT count(*) FROM pg_stat_activity WHERE " + predicate) == "0")

            lock_process.stdin.write("COMMIT;\n\\q\n")
            lock_process.stdin.flush()
            stdout, stderr = lock_process.communicate(timeout=min(10, deadline - time.monotonic()))
            assert lock_process.returncode == 0, "lock release failed: INVALID: " + stderr
            reader.join(timeout=1)
            assert not reader.is_alive(), "lock reader remained active: INVALID"
            lock_process = None
            if arm == "normal":
                assert run("wait", container) == "0", "normal restore did not complete"
            wait_for(lambda: sql("postgres", "SELECT count(*) FROM pg_stat_activity WHERE " + predicate) == "0")
            table_exists = sql(database, "SELECT to_regclass('public.restore_fence_probe') IS NOT NULL") == "t"
            rows = int(sql(database, "SELECT count(*) FROM public.restore_fence_probe")) if table_exists else None
            if arm == "normal":
                assert table_exists and rows == 1 and sql(database, "SELECT id FROM public.restore_fence_probe") == "7"
            observations.append(dict(arm=arm, blocked_query="COPY", rows_before_intervention=baseline_rows,
                                     backend_survived_client_removal=backend_after_removal,
                                     backend_termination_result=terminated, table_exists_after_unlock=table_exists, rows=rows))
    finally:
        try:
            if lock_process is not None:
                lock_process.kill()
                lock_process.communicate(timeout=10)
        finally:
            # Every destructive cleanup target belongs to this unpredictable,
            # experiment-owned project. Never run a global prune or fixed project.
            ids = run("ps", "--all", "--quiet", "--no-trunc", "--filter", "label=com.docker.compose.project=" + project, cleanup=True).split()
            for container in ids:
                label = run("inspect", "--format", '{{index .Config.Labels "com.docker.compose.project"}}', container, cleanup=True)
                assert label == project, "refuse unowned cleanup"
            run(*compose, "down", "--volumes", "--remove-orphans", cleanup=True)
            assert not run("ps", "--all", "--quiet", "--filter", "label=com.docker.compose.project=" + project, cleanup=True)
            assert not run("network", "ls", "--quiet", "--filter", "label=com.docker.compose.project=" + project, cleanup=True)
            assert not run("volume", "ls", "--quiet", "--filter", "label=com.docker.compose.project=" + project, cleanup=True)
            shutil.rmtree(work)
    # Cleanup failure prevents publication of a complete observation.
    print(json.dumps(dict(project=project, objective_completed=True, guard_exhausted=False,
                          observations=observations), indent=2))


if __name__ == "__main__":
    try:
        main()
    except Exception as error:
        guard = isinstance(error, (subprocess.TimeoutExpired, queue.Empty)) or "guard exhausted" in str(error)
        print(json.dumps(dict(objective_completed=False, guard_exhausted=guard,
                              error_class="invalid_experiment")), flush=True)
        raise
