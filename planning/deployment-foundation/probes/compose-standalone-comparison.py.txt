def require(ok, message):
    if not ok:
        raise RuntimeError(message)


class JSONNumber(str):
    """Exact JSON number token, distinct from a JSON string or boolean."""


def model_object(pairs):
    result = {}
    for key, value in pairs:
        require(key not in result, "invalid JSON model: duplicate key")
        result[key] = value
    return result


def reject_constant(token):
    raise RuntimeError("invalid JSON model: nonfinite constant " + token)


def parse_model(text):
    return json.loads(text, parse_int=JSONNumber, parse_float=JSONNumber,
                      parse_constant=reject_constant, object_pairs_hook=model_object)


def equal_models(left, right):
    # Only object-key ordering is ignored. Number tokens retain their full text.
    require(type(left) is type(right), "complete parsed model mismatch")
    if isinstance(left, dict):
        require(left.keys() == right.keys(), "complete parsed model mismatch")
        for key in left:
            equal_models(left[key], right[key])
    elif isinstance(left, list):
        require(len(left) == len(right), "complete parsed model mismatch")
        for a, b in zip(left, right):
            equal_models(a, b)
    else:
        require(type(left) in (str, JSONNumber, bool, type(None)),
                "invalid JSON model scalar")
        require(left == right, "complete parsed model mismatch")


def models_must_differ(left, right):
    try:
        equal_models(left, right)
    except RuntimeError as error:
        require(str(error) == "complete parsed model mismatch", "unexpected comparison error")
    else:
        raise RuntimeError("model comparator accepted a changed value/type")


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", required=True)
    parser.add_argument("--configuration-input-mode", choices=("dotenv", "process_environment", "process_environment_absent_secrets"), default="dotenv")
    args = parser.parse_args()
    root = pathlib.Path.cwd()
    output = pathlib.Path(args.output).resolve()
    require(not output.exists() and output.parent.is_dir(), "fresh output required")
    require((root / "Makefile").is_file(), "repository root required")
    tools = root / ".cache/worker-cli-tools"
    metadata = json.loads((tools / "tools.json").read_text())
    for entry in metadata["binaries"]:
        require(hashlib.sha256((root / entry["path"]).read_bytes()).hexdigest() == entry["sha256"], "cached binary changed")
    compose_hash = metadata["binaries"][1]["sha256"]
    cli_hash = metadata["binaries"][0]["sha256"]
    project = "cc-standalone-" + secrets.token_hex(8)
    work = pathlib.Path(tempfile.mkdtemp(prefix=project + "-"))
    work.chmod(0o755)
    containers = []
    attempted_create = False
    completed = False
    deadline = None

    def run(*command, cleanup=False):
        now = time.monotonic()
        until = now + CLEANUP_SECONDS if cleanup else min(deadline, now + COMMAND_SECONDS)
        require(until > now, "outer guard: INVALID")
        reply = subprocess.run(["docker", *command], capture_output=True, text=True, timeout=until-now)
        require(time.monotonic() < until, "outer command guard: INVALID")
        require(reply.returncode == 0, "owned Docker command failed: " + command[0] + " " + reply.stderr)
        return reply.stdout.strip()

    def owned(container, mode, cleanup=False):
        actual = run("inspect", "--format", '{{.Id}} {{.Image}} {{index .Config.Labels "com.docker.compose.project"}} {{index .Config.Labels "com.docker.compose.service"}}', container, cleanup=cleanup)
        require(actual == " ".join((container, PYTHON, project, mode)), "container ownership mismatch")

    try:
        images = {name: "research.invalid/" + name + "@sha256:" + str(index)*64 for index, name in enumerate(("alertmanager", "caddy", "gameserver", "node-exporter", "postgres", "prometheus"), 1)}
        render = subprocess.run(["make", "render-release-compose", "RELEASE_COMPOSE_OUTPUT=" + str(work / "base.yml"), *[name.upper().replace("-", "_") + "_IMAGE=" + image for name, image in images.items()]], capture_output=True, text=True, timeout=BUILD_SECONDS)
        require(render.returncode == 0, "actual release renderer failed: " + render.stdout + render.stderr)
        shutil.copyfile(root / "deployment/compose.rotation.template.yml", work / "rotation.yml")
        public_env = (root / "deployment/.env.example").read_text()
        public_env += "\nCLOUD_CLICKER_JWT_PREVIOUS_ID=jwt-previous-public\nCLOUD_CLICKER_BOOTSTRAP_PREVIOUS_ID=bootstrap-previous-public\nCLOUD_CLICKER_JWT_PREVIOUS_SECRET_FILE=./secrets/jwt-previous\nCLOUD_CLICKER_BOOTSTRAP_PREVIOUS_SECRET_FILE=./secrets/bootstrap-previous\n"
        (work / "public.env").write_text(public_env)
        (work / "empty.env").write_text("# deliberately missing required values\n")
        (work / "minimal.yml").write_text("services:\n  probe:\n    image: " + PYTHON + "\n    network_mode: none\n    read_only: true\n")
        (work / "invalid.yml").write_text("services: [\n")
        (work / "wrong-type.yml").write_text("services: []\n")
        (work / "docker-config").mkdir()
        (work / "docker-config/config.json").write_text(json.dumps(dict(auths={}, cliPluginsExtraDirs=["/tools/plugins"])))
        (work / "direct-docker-config").mkdir()
        (work / "direct-docker-config/config.json").write_text(json.dumps(dict(auths={})))
        (work / "secrets").mkdir()
        for name in ("database-url", "postgres-password", "jwt-current", "bootstrap-current", "cursor-current", "alertmanager.yml", "jwt-previous", "bootstrap-previous"):
            (work / "secrets" / name).write_text("public synthetic test input, never an operator secret\n")
        cases = [
            dict(id="minimal", env="public.env", files=["minimal.yml"], success=True),
            dict(id="production_base", env="public.env", files=["base.yml"], success=True),
            dict(id="production_rotation", env="public.env", files=["base.yml", "rotation.yml"], success=True),
            dict(id="invalid_yaml", env="public.env", files=["invalid.yml"], success=False),
            dict(id="wrong_type", env="public.env", files=["wrong-type.yml"], success=False),
            dict(id="required_env_missing", env="empty.env", files=["base.yml"], success=False),
            dict(id="input_missing", env="public.env", files=["missing.yml"], success=False),
        ]
        (work / "cases.json").write_text(json.dumps(cases))
        for path in work.rglob("*"):
            path.chmod(0o755 if path.is_dir() else 0o644)
        fixture_hashes = {str(p.relative_to(work)): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(work.rglob("*")) if p.is_file()}
        deadline = time.monotonic() + WORK_SECONDS
        engine = run("version", "--format", "{{.Server.Version}} {{.Server.Os}}/{{.Server.Arch}}")
        image = json.loads(run("image", "inspect", PYTHON, "--format", '{"id":{{json .Id}},"os":{{json .Os}},"arch":{{json .Architecture}}}'))
        require(image == dict(id=PYTHON, os="linux", arch="arm64"), "unmeasured fixture image platform")
        results = {}
        profiles = {}
        print(json.dumps(dict(phase="started", project=project, cases=len(cases), modes=4)), flush=True)
        for mode in ("wrapped", "direct", "missing", "swapped"):
            mounts = ["type=bind,source=" + str(work) + ",target=/fixture,readonly"]
            destinations = ["/fixture"]
            if mode == "wrapped":
                mounts += ["type=bind,source=" + str(tools / "docker") + ",target=/tools/docker,readonly", "type=bind,source=" + str(tools / "plugins/docker-compose") + ",target=/tools/plugins/docker-compose,readonly"]
                destinations += ["/tools/docker", "/tools/plugins/docker-compose"]
            elif mode != "missing":
                source = tools / ("docker" if mode == "swapped" else "plugins/docker-compose")
                mounts += ["type=bind,source=" + str(source) + ",target=/tools/docker-compose,readonly"]
                destinations += ["/tools/docker-compose"]
            argv = ["create", "--pull=never", "--name", project + "-" + mode, "--label", "com.docker.compose.project=" + project, "--label", "com.docker.compose.service=" + mode, "--network", "none", "--ipc", "private", "--read-only", "--user", "65532:65532", "--cap-drop", "ALL", "--security-opt", "no-new-privileges:true"]
            for mount in mounts:
                argv += ["--mount", mount]
            attempted_create = True
            container = run(*argv, "--entrypoint", "python3", PYTHON, "-c", WORKER, mode, compose_hash, cli_hash, args.configuration_input_mode)
            containers.append((container, mode))
            owned(container, mode)
            raw = json.loads(run("inspect", container))[0]
            selected = dict(user=raw["Config"]["User"], network=raw["HostConfig"]["NetworkMode"], ipc=raw["HostConfig"]["IpcMode"], pid=raw["HostConfig"]["PidMode"], readonly=raw["HostConfig"]["ReadonlyRootfs"], privileged=raw["HostConfig"]["Privileged"], caps=raw["HostConfig"]["CapDrop"], security=raw["HostConfig"]["SecurityOpt"], mounts=sorted([dict(path=m["Destination"], type=m["Type"], rw=m["RW"]) for m in raw["Mounts"]], key=lambda x:x["path"]))
            expected = dict(user="65532:65532", network="none", ipc="private", pid="", readonly=True, privileged=False, caps=["ALL"], security=["no-new-privileges:true"], mounts=[dict(path=p, type="bind", rw=False) for p in sorted(destinations)])
            require(selected == expected, "actual fixture boundary mismatch")
            profiles[mode] = selected
            run("start", container)
            while run("inspect", "--format", "{{.State.Running}}", container) == "true":
                require(time.monotonic() < deadline, "worker observation guard: INVALID")
                time.sleep(0.1)
            exit_code = run("wait", container)
            if exit_code != "0":
                error_lines = [line for line in run("logs", container).splitlines() if line.startswith(("RuntimeError:", "subprocess.TimeoutExpired:"))]
                raise RuntimeError("worker incomplete: " + mode + " exit=" + exit_code + " " + " ".join(error_lines))
            observed = json.loads(run("logs", container))
            require(observed["objective_completed"] is True, "worker objective incomplete")
            require(observed['configuration_input_mode'] == args.configuration_input_mode, 'input-mode observation mismatch')
            if mode in ("wrapped", "direct"):
                observed['models'] = {name: parse_model(text) for name, text in observed['models'].items()}
            results[mode] = observed
        for case in cases:
            if case["success"]:
                equal_models(results["wrapped"]["models"][case["id"]], results["direct"]["models"][case["id"]])
        production = results["direct"]["models"]["production_base"]
        require(len(production["services"]) == 7 and len(production["networks"]) == 4 and len(production["volumes"]) == 5 and production["services"]["gameserver"]["read_only"] is True, "production model did not exercise real boundaries")
        for wrong in (False, parse_model("1")):
            altered = copy.deepcopy(production)
            altered["services"]["gameserver"]["read_only"] = wrong
            models_must_differ(production, altered)
        equal_models(production, dict(reversed(list(production.items()))))
        for left, right in (("9007199254740992.0", "9007199254740993.0"),
                            ("1", '"1"'), ("1", "1.0")):
            models_must_differ(parse_model(left), parse_model(right))
        for invalid in ('{"value":1,"value":2}', '{"value":NaN}'):
            try:
                parse_model(invalid)
            except RuntimeError as error:
                require(str(error).startswith("invalid JSON model:"), "unexpected parse error")
            else:
                raise RuntimeError("ambiguous/nonfinite model accepted")
        byte_equal_pairs = sum(a['stdout_sha256'] == b['stdout_sha256']
                               for a, b in zip(results['wrapped']['rows'], results['direct']['rows'])
                               if a['case'] in results['direct']['models'])
        for mode in ("wrapped", "direct"):
            results[mode].pop("models")
        require(time.monotonic() < deadline, "work guard: INVALID")
        observation = dict(schema_version=4, configuration_input_mode=args.configuration_input_mode, objective_completed=True, research_only=True, project=project, engine=engine, fixture_image=image, binary_inputs=metadata["binaries"], fixture_hashes=fixture_hashes, cases=cases, results=results, profiles=profiles, model_comparisons=sum(c['success'] for c in cases), byte_equal_pairs=byte_equal_pairs, policy_field_change_refused=True, boolean_numeric_change_refused=True, object_key_order_accepted=True, number_precision_change_refused=True, number_string_change_refused=True, number_token_spelling_change_refused=True, duplicate_json_keys_refused=True, nonfinite_json_refused=True, source_sha256=hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(), boundary_source_sha256=hashlib.sha256(BOUNDARY.read_bytes()).hexdigest(), guards=dict(work=WORK_SECONDS, command=COMMAND_SECONDS, cleanup=CLEANUP_SECONDS, build=BUILD_SECONDS), limitations=["Desktop ARM64 fixture with AMD64 tool execution; not supported native Linux/amd64", "Configuration compilation only; no service applied or native supervisor proved", "Synthetic public configuration and placeholder image references; no release artifact eligibility", "Public models traverse Docker stdout/logs; only final observation excludes models", "No private-model transport or memory-only handling guarantee", "No operator Docker config/credentials/network/Engine socket available inside workers", "No rights/artifact/profile adoption or acceptance inferred"])
        completed = True
    finally:
        for container, mode in reversed(containers):
            owned(container, mode, cleanup=True)
            run("rm", "--force", "--volumes", container, cleanup=True)
        leftover = run("container", "ls", "--all", "--quiet", "--no-trunc", "--filter", "label=com.docker.compose.project=" + project, cleanup=True) if attempted_create else ""
        for container in leftover.splitlines():
            mode = run("inspect", "--format", '{{index .Config.Labels "com.docker.compose.service"}}', container, cleanup=True)
            require(mode in ("wrapped", "direct", "missing", "swapped"), "unknown leftover fixture role")
            owned(container, mode, cleanup=True)
            run("rm", "--force", "--volumes", container, cleanup=True)
        if attempted_create:
            require(run("container", "ls", "--all", "--quiet", "--filter", "label=com.docker.compose.project=" + project, cleanup=True) == "", "owned cleanup incomplete")
        if completed:
            shutil.rmtree(work)
        else:
            print(json.dumps(dict(objective_completed=False, private_work=str(work))), flush=True)
    observation["cleanup_complete"] = True
    with output.open("x") as handle:
        json.dump(observation, handle, sort_keys=True, indent=2)
        handle.write("\n")
    print(json.dumps(dict(objective_completed=True, cleanup_complete=True, output=str(output), comparisons=observation['model_comparisons'], failure_cases=sum(not c['success'] for c in cases)*2, binary_controls=2, policy_field_change_refused=True, boolean_numeric_change_refused=True, object_key_order_accepted=True)), flush=True)


if __name__ == "__main__":
    main()
